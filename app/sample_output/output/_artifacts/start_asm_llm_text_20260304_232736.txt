;******************************************************************************
; @file    start.s
; @brief   Cortex-R4 startup code for TI RM46 (TI ARM CGT assembler syntax)
;
; Exception vectors are executable ARM instructions (not address words).
; Stack pointers for all exception modes are initialized before entering C.
;******************************************************************************

    .sect ".intvecs"
    .align 4
    .global Reset_Handler

;------------------------------------------------------------------------------
; Exception Vector Table (Cortex-R style: executable instructions)
;------------------------------------------------------------------------------
Reset_Vector:
    B       Reset_Handler               ; Reset
    B       Undef_Handler               ; Undefined Instruction
    B       SVC_Handler                 ; Supervisor Call
    B       Prefetch_Abort_Handler      ; Prefetch Abort
    B       Data_Abort_Handler          ; Data Abort
    B       Phantom_Handler             ; Reserved/Phantom
    LDR     PC, [PC, #-0x1B0]           ; IRQ (VIM-style)
    LDR     PC, [PC, #-0x1B0]           ; FIQ (VIM-style)

;------------------------------------------------------------------------------
; Reset Handler
; 1. Initialize stack pointers for all exception modes
; 2. Return to SVC mode
; 3. Call Reset_Handler_C
; 4. Loop forever if Reset_Handler_C returns
;------------------------------------------------------------------------------
    .sect ".text"
    .align 4
    .ref Reset_Handler_C
    .ref end_of_stack

Reset_Handler:
    ; Read current CPSR
    MRS     R0, CPSR

    ; Switch to FIQ mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x11           ; FIQ mode (0x11)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Switch to IRQ mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x12           ; IRQ mode (0x12)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Switch to Abort mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x17           ; Abort mode (0x17)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Switch to Undefined mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x1B           ; Undefined mode (0x1B)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Switch to System mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x1F           ; System mode (0x1F)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Switch to Supervisor mode and set SP
    BIC     R1, R0, #0x1F           ; Clear mode bits
    ORR     R1, R1, #0x13           ; Supervisor mode (0x13)
    MSR     CPSR_c, R1
    LDR     SP, stack_addr

    ; Branch to C entry point
    BL      Reset_Handler_C

Reset_Loop:
    B       Reset_Loop              ; Loop forever if Reset_Handler_C returns

;------------------------------------------------------------------------------
; Exception Handlers (minimal infinite loops)
;------------------------------------------------------------------------------
Undef_Handler:
    B       Undef_Handler

SVC_Handler:
    B       SVC_Handler

Prefetch_Abort_Handler:
    B       Prefetch_Abort_Handler

Data_Abort_Handler:
    B       Data_Abort_Handler

Phantom_Handler:
    B       Phantom_Handler

;------------------------------------------------------------------------------
; Stack address literal pool
;------------------------------------------------------------------------------
    .align 4
stack_addr:
    .long end_of_stack

;------------------------------------------------------------------------------
; End of start.s
;------------------------------------------------------------------------------
