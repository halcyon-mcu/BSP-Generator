/**
 * @file vim_driver.h
 * @brief VIM (Vectored Interrupt Manager) driver for TI RM46
 */

#ifndef VIM_DRIVER_H
#define VIM_DRIVER_H

#include <stdint.h>

/**
 * @defgroup BSP_VIM VIM Driver
 * @brief Vectored Interrupt Manager (VIM) driver providing interrupt controller initialization and channel management.
 * @{
 */

/**
 * @brief VIM ISR function pointer type
 * @ingroup BSP_VIM
 *
 * User interrupt service routines must match this signature.
 */
typedef void (*vim_isr_t)(void);

/**
 * @brief Initialize the VIM for vectored interrupt mode
 * @ingroup BSP_VIM
 *
 * This function:
 * - Enables VIM parity checking if supported
 * - Initializes all vector table entries to a default handler
 * - Does NOT enable any interrupt channels by default
 *
 * @note Must be called before registering ISRs or enabling channels
 * @warning Does not configure system-level interrupt enable (CPSR I/F bits)
 */
void vim_init(void);

/**
 * @brief Register an ISR for a specific VIM channel
 * @ingroup BSP_VIM
 *
 * Writes the ISR address into the VIM vector table entry corresponding to the given channel.
 *
 * @param[in] channel_id VIM channel number (valid range: 2..126, excluding reserved channels)
 * @param[in] isr Pointer to the interrupt service routine
 * @return 0 on success, -1 if channel_id is invalid or reserved
 *
 * @note Channel 0 is phantom and channel 127 is reserved
 * @note This does not enable the channel; call vim_enable_channel() separately
 */
int vim_register_isr(uint32_t channel_id, vim_isr_t isr);

/**
 * @brief Enable a VIM interrupt channel
 * @ingroup BSP_VIM
 *
 * Sets the corresponding bit in the VIM REQENASET registers to enable the channel.
 *
 * @param[in] channel_id VIM channel number (valid range: 2..126, excluding reserved channels)
 * @return 0 on success, -1 if channel_id is invalid or reserved
 *
 * @note An ISR should be registered with vim_register_isr() before enabling the channel
 */
int vim_enable_channel(uint32_t channel_id);

/**
 * @brief Disable a VIM interrupt channel
 * @ingroup BSP_VIM
 *
 * Sets the corresponding bit in the VIM REQENACLR registers to disable the channel.
 *
 * @param[in] channel_id VIM channel number (valid range: 2..126, excluding reserved channels)
 * @return 0 on success, -1 if channel_id is invalid or reserved
 */
int vim_disable_channel(uint32_t channel_id);

/**
 * @brief Configure a VIM channel as FIQ or IRQ
 * @ingroup BSP_VIM
 *
 * Routes the specified channel to either FIQ or IRQ using the FIRQPR registers.
 *
 * @param[in] channel_id VIM channel number (valid range: 2..126, excluding reserved channels)
 * @param[in] enable_fiq 1 to route to FIQ, 0 to route to IRQ
 * @return 0 on success, -1 if channel_id is invalid or reserved
 *
 * @note Channels 0 and 1 are hardwired to FIQ and cannot be changed
 */
int vim_set_fiq(uint32_t channel_id, int enable_fiq);

/** @} */

#endif /* VIM_DRIVER_H */

/**
 * @file vim_driver.c
 * @brief VIM (Vectored Interrupt Manager) driver implementation for TI RM46
 */

#include "vim_driver.h"
#include <stdint.h>

/* Register access macro */
#define REG32(addr) (*(volatile uint32_t *)(addr))

/* VIM register base address */
/* [prov] regs.yaml:VIM.base_address */
#define VIM_BASE (0xFFFFFE00u)

/* VIM register offsets */
/* [prov] regs.yaml:VIM.FIRQPR0 */
#define VIM_FIRQPR0_OFFSET (0x10u)
/* [prov] regs.yaml:VIM.FIRQPR1 */
#define VIM_FIRQPR1_OFFSET (0x14u)
/* [prov] regs.yaml:VIM.FIRQPR2 */
#define VIM_FIRQPR2_OFFSET (0x18u)
/* [prov] regs.yaml:VIM.FIRQPR3 */
#define VIM_FIRQPR3_OFFSET (0x1Cu)

/* [prov] regs.yaml:VIM.REQENASET0 */
#define VIM_REQENASET0_OFFSET (0x30u)
/* [prov] regs.yaml:VIM.REQENASET1 */
#define VIM_REQENASET1_OFFSET (0x34u)
/* [prov] regs.yaml:VIM.REQENASET2 */
#define VIM_REQENASET2_OFFSET (0x38u)
/* [prov] regs.yaml:VIM.REQENASET3 */
#define VIM_REQENASET3_OFFSET (0x3Cu)

/* [prov] regs.yaml:VIM.REQENACLR0 */
#define VIM_REQENACLR0_OFFSET (0x40u)
/* [prov] regs.yaml:VIM.REQENACLR1 */
#define VIM_REQENACLR1_OFFSET (0x44u)
/* [prov] regs.yaml:VIM.REQENACLR2 */
#define VIM_REQENACLR2_OFFSET (0x48u)
/* [prov] regs.yaml:VIM.REQENACLR3 */
#define VIM_REQENACLR3_OFFSET (0x4Cu)

/* [prov] regs.yaml:VIM.PARCTL */
#define VIM_PARCTL_OFFSET (0xF0u)
/* [prov] regs.yaml:VIM.FBPARERR */
#define VIM_FBPARERR_OFFSET (0xF8u)

/* [prov] regs.yaml:VIM.PARCTL.PARENA */
#define VIM_PARCTL_ENABLE_VALUE (0x0Au)

/* VIM vector table constants */
/* [prov] soc.yaml:VIM.x-ext.vector_table.base_address */
#define VIM_VECTOR_TABLE_BASE (0xFFF82000u)
/* [prov] soc.yaml:VIM.x-ext.vector_table.entries */
#define VIM_VECTOR_TABLE_ENTRIES (128u)
/* [prov] soc.yaml:VIM.x-ext.vector_table.entry_size_bytes */
#define VIM_VECTOR_TABLE_ENTRY_SIZE (4u)
/* [prov] soc.yaml:VIM.x-ext.vector_table.phantom_entry */
#define VIM_PHANTOM_ENTRY (0u)
/* [prov] soc.yaml:VIM.x-ext.vector_table.reserved_channels */
#define VIM_RESERVED_CHANNEL (127u)

/* Forward declaration of default ISR */
static void vim_default_isr(void);

/**
 * @brief Default ISR for unhandled interrupts
 *
 * This function is called for any interrupt channel that has not had
 * a specific ISR registered. It enters an infinite loop.
 */
static void vim_default_isr(void)
{
    /* Infinite loop for unhandled interrupts */
    while (1)
    {
        /* User should override vector table entry or handle this condition */
    }
}

void vim_init(void)
{
    uint32_t i;
    volatile uint32_t *vec_table;

    /* [prov] regs.yaml:VIM.FBPARERR */
    /* Set fall-back parity error address to default handler */
    REG32(VIM_BASE + VIM_FBPARERR_OFFSET) = (uint32_t)vim_default_isr;

    /* [prov] regs.yaml:VIM.PARCTL */
    /* Enable VIM parity checking (write 0xA to PARENA field) */
    REG32(VIM_BASE + VIM_PARCTL_OFFSET) = VIM_PARCTL_ENABLE_VALUE;

    /* [prov] soc.yaml:VIM.x-ext.vector_table (base/entries) */
    /* Initialize all vector table entries to default handler */
    vec_table = (volatile uint32_t *)VIM_VECTOR_TABLE_BASE;
    for (i = 0; i < VIM_VECTOR_TABLE_ENTRIES; i++)
    {
        vec_table[i] = (uint32_t)vim_default_isr;
    }

    /* Do not enable any channels by default */
}

int vim_register_isr(uint32_t channel_id, vim_isr_t isr)
{
    volatile uint32_t *vec_entry;
    uint32_t entry_index;

    /* Validate channel_id: must be in range 2..126, excluding reserved channel 127 */
    if ((channel_id < 2u) || (channel_id >= (VIM_VECTOR_TABLE_ENTRIES - 1u)) || (channel_id == VIM_RESERVED_CHANNEL))
    {
        return -1;
    }

    /* [prov] soc.yaml:VIM.x-ext.vector_table (base/entries) */
    /* Calculate vector table entry: channel N uses entry (N + 1) due to phantom entry 0 */
    entry_index = channel_id + 1u;
    vec_entry = (volatile uint32_t *)(VIM_VECTOR_TABLE_BASE + (entry_index * VIM_VECTOR_TABLE_ENTRY_SIZE));

    /* Write ISR address to vector table */
    *vec_entry = (uint32_t)isr;

    return 0;
}

int vim_enable_channel(uint32_t channel_id)
{
    uint32_t reg_index;
    uint32_t bit_offset;
    uint32_t reg_offset;

    /* Validate channel_id */
    if ((channel_id < 2u) || (channel_id >= (VIM_VECTOR_TABLE_ENTRIES - 1u)) || (channel_id == VIM_RESERVED_CHANNEL))
    {
        return -1;
    }

    /* Determine which REQENASET register and bit position */
    reg_index = channel_id / 32u;
    bit_offset = channel_id % 32u;

    /* Select the appropriate REQENASET register */
    switch (reg_index)
    {
        case 0:
            /* [prov] regs.yaml:VIM.REQENASET0 */
            reg_offset = VIM_REQENASET0_OFFSET;
            break;
        case 1:
            /* [prov] regs.yaml:VIM.REQENASET1 */
            reg_offset = VIM_REQENASET1_OFFSET;
            break;
        case 2:
            /* [prov] regs.yaml:VIM.REQENASET2 */
            reg_offset = VIM_REQENASET2_OFFSET;
            break;
        case 3:
            /* [prov] regs.yaml:VIM.REQENASET3 */
            reg_offset = VIM_REQENASET3_OFFSET;
            break;
        default:
            return -1;
    }

    /* Write 1 to the corresponding bit to enable the channel */
    REG32(VIM_BASE + reg_offset) = (1u << bit_offset);

    return 0;
}

int vim_disable_channel(uint32_t channel_id)
{
    uint32_t reg_index;
    uint32_t bit_offset;
    uint32_t reg_offset;

    /* Validate channel_id */
    if ((channel_id < 2u) || (channel_id >= (VIM_VECTOR_TABLE_ENTRIES - 1u)) || (channel_id == VIM_RESERVED_CHANNEL))
    {
        return -1;
    }

    /* Determine which REQENACLR register and bit position */
    reg_index = channel_id / 32u;
    bit_offset = channel_id % 32u;

    /* Select the appropriate REQENACLR register */
    switch (reg_index)
    {
        case 0:
            /* [prov] regs.yaml:VIM.REQENACLR0 */
            reg_offset = VIM_REQENACLR0_OFFSET;
            break;
        case 1:
            /* [prov] regs.yaml:VIM.REQENACLR1 */
            reg_offset = VIM_REQENACLR1_OFFSET;
            break;
        case 2:
            /* [prov] regs.yaml:VIM.REQENACLR2 */
            reg_offset = VIM_REQENACLR2_OFFSET;
            break;
        case 3:
            /* [prov] regs.yaml:VIM.REQENACLR3 */
            reg_offset = VIM_REQENACLR3_OFFSET;
            break;
        default:
            return -1;
    }

    /* Write 1 to the corresponding bit to disable the channel */
    REG32(VIM_BASE + reg_offset) = (1u << bit_offset);

    return 0;
}

int vim_set_fiq(uint32_t channel_id, int enable_fiq)
{
    uint32_t reg_index;
    uint32_t bit_offset;
    uint32_t reg_offset;
    uint32_t reg_value;

    /* Validate channel_id: channels 0 and 1 are hardwired to FIQ */
    if ((channel_id < 2u) || (channel_id >= (VIM_VECTOR_TABLE_ENTRIES - 1u)) || (channel_id == VIM_RESERVED_CHANNEL))
    {
        return -1;
    }

    /* Determine which FIRQPR register and bit position */
    reg_index = channel_id / 32u;
    bit_offset = channel_id % 32u;

    /* Select the appropriate FIRQPR register */
    switch (reg_index)
    {
        case 0:
            /* [prov] regs.yaml:VIM.FIRQPR0 */
            reg_offset = VIM_FIRQPR0_OFFSET;
            break;
        case 1:
            /* [prov] regs.yaml:VIM.FIRQPR1 */
            reg_offset = VIM_FIRQPR1_OFFSET;
            break;
        case 2:
            /* [prov] regs.yaml:VIM.FIRQPR2 */
            reg_offset = VIM_FIRQPR2_OFFSET;
            break;
        case 3:
            /* [prov] regs.yaml:VIM.FIRQPR3 */
            reg_offset = VIM_FIRQPR3_OFFSET;
            break;
        default:
            return -1;
    }

    /* Read current value */
    reg_value = REG32(VIM_BASE + reg_offset);

    /* Modify the bit: 1 = FIQ, 0 = IRQ */
    if (enable_fiq)
    {
        reg_value |= (1u << bit_offset);
    }
    else
    {
        reg_value &= ~(1u << bit_offset);
    }

    /* Write back modified value */
    REG32(VIM_BASE + reg_offset) = reg_value;

    return 0;
}
