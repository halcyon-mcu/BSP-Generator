273

276 }
277
278 /* Find pin in lookup table */
279 mapping = NULL;
280 for (i = 0U; i < PIN_MAPPING_COUNT; i++) {

283 break;

288 return IOMM_STATUS_INVALID_PIN;

290

292 pinmmr_reg = &iommREG->PINMMR0 + mapping->pinmmr_reg;
293

295 reg_value = *pinmmr_reg;
