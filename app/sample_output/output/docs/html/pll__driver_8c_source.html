642 if (status == (pll_lock_status_t*)0) {
643 return;
644 }
645
646 csvstat = SYSREG->CSVSTAT;
647 glbstat = SYSREG->GLBSTAT;
648
649 status->pll1_locked = ((csvstat & SYSTEM_CSVSTAT_CLKSR1V) != 0U);
650 status->pll2_locked = ((csvstat & SYSTEM_CSVSTAT_CLKSR6V) != 0U);
651 status->clock_slip_detected = ((glbstat & (SYSTEM_GLBSTAT_RFSLIP | SYSTEM_GLBSTAT_FBSLIP)) != 0U);
652}

654/**
655 * @brief Poll for PLL lock with timeout
656 * @details Waits for specified PLL to achieve lock state.
657 *
658 * @param pll PLL instance to wait for
659 * @param timeout_us Timeout in microseconds (0 = infinite)
660 * @return pll_status_t PLL_STATUS_OK when locked, PLL_STATUS_TIMEOUT on timeout
661 */
662pll_status_t PLL_WaitForLock(pll_instance_t pll, uint32_t timeout_us)
663{
664 uint32_t timeout;
665
666 /* Convert timeout to iteration count (rough approximation) */
667 timeout = (timeout_us == 0U) ? 0xFFFFFFFFU : (timeout_us / 10U);
668 if (timeout == 0U) {
669 timeout = 1U;
670 }
671
672 if (pll == PLL_INSTANCE_1) {
673 while (((SYSREG->CSVSTAT & SYSTEM_CSVSTAT_CLKSR1V) == 0U) && (timeout > 0U)) {
674 timeout--;
675 }
676 } else if (pll == PLL_INSTANCE_2) {
677 while (((SYSREG->CSVSTAT & SYSTEM_CSVSTAT_CLKSR6V) == 0U) && (timeout > 0U)) {
678 timeout--;
679 }
680 } else {
681 return PLL_STATUS_INVALID_PARAM;
682 }
683
684 return (timeout > 0U) ? PLL_STATUS_OK : PLL_STATUS_TIMEOUT;
685}

687/**
688 * @brief Enable the specified PLL instance
689 * @details Enables PLL and waits for lock before returning.
690 *
691 * @param pll PLL instance to enable
692 * @return pll_status_t Status code
693 */
694pll_status_t PLL_EnablePLL(pll_instance_t pll)
695{
696 if (pll == PLL_INSTANCE_1) {
697 SYSREG->CSDISCLR = SYSTEM_CSDISCLR_CLRCLKSR1OFF;
698 return wait_for_pll1_lock();
699 } else if (pll == PLL_INSTANCE_2) {

701 return wait_for_pll2_lock();
702 } else {
703 return PLL_STATUS_INVALID_PARAM;
704 }
705}

747/**

751 * @param domain Clock domain to configure
752 * @param source_id Source ID value

754 */
755pll_status_t PLL_SelectClockSource(clock_domain_t domain, uint32_t source_id)
756{
757 uint32_t reg_value;
758
759 switch (domain) {
760 case CLOCKDOMAIN_GCLK:
761 case CLOCKDOMAIN_HCLK:
762 /* Configure GHVSRC */
763 reg_value = SYSREG->GHVSRC;
764 reg_value &= ~SYSTEM_GHVSRC_GHVSRC_MASK;

766 SYSREG->GHVSRC = reg_value;

769 case CLOCKDOMAIN_VCLKA1:
770 /* Configure VCLKASRC */
771 reg_value = SYSREG->VCLKASRC;
772 reg_value &= ~SYSTEM_VCLKASRC_VCLKA1S_MASK;
773 reg_value |= (source_id << SYSTEM_VCLKASRC_VCLKA1S_SHIFT);
774 SYSREG->VCLKASRC = reg_value;
775 break;
776
777 case CLOCKDOMAIN_RTICLK:
778 /* Configure RCLKSRC */
779 reg_value = SYSREG->RCLKSRC;
780 reg_value &= ~SYSTEM_RCLKSRC_RTI1SRC_MASK;
781 reg_value |= (source_id << SYSTEM_RCLKSRC_RTI1SRC_SHIFT);
782 SYSREG->RCLKSRC = reg_value;
783 break;
784
785 case CLOCKDOMAIN_VCLKA3:
786 /* Configure VCLKACON1 VCLKA3S */
787 reg_value = SYSREG2->VCLKACON1;
788 reg_value &= ~SYSTEM2_VCLKACON1_VCLKA3S_MASK;
789 reg_value |= (source_id << SYSTEM2_VCLKACON1_VCLKA3S_SHIFT);
790 SYSREG2->VCLKACON1 = reg_value;
791 break;
792
793 case CLOCKDOMAIN_VCLKA4:
794 /* Configure VCLKACON1 VCLKA4S */
795 reg_value = SYSREG2->VCLKACON1;
796 reg_value &= ~SYSTEM2_VCLKACON1_VCLKA4S_MASK;
797 reg_value |= (source_id << SYSTEM2_VCLKACON1_VCLKA4S_SHIFT);
798 SYSREG2->VCLKACON1 = reg_value;
799 break;
800
801 default:
802 return PLL_STATUS_INVALID_PARAM;
803 }
804
805 return PLL_STATUS_OK;
806}
