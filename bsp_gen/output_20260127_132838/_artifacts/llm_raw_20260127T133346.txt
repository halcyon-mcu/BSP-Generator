/* Helper: wait for a flag bit to be set, with timeout */
static int sci_wait_flag(uint32_t flag_bit, uint32_t timeout)
{
    uint32_t i;
    for (i = 0; i < timeout; i++) {
        /* [prov] regs.yaml:SCI.SCIFLR */
        if (REG32(SCIFLR) & (1u << flag_bit)) {
            return 0;
        }
    }
    return -1;
}

void sci_init(void)
{
    /* Enable VCLK clock */
    clock_enable(CLOCKREF_VCLK);

    /* [prov] regs.yaml:SCI.SCIGCR0 */
    /* Step 1: Set RESET bit to reset the module */
    REG32(SCIGCR0) = SCIGCR0_INIT_VALUE;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    /* Step 2: Clear SWnRST bit (hold in reset) */
    REG32(SCIGCR1) &= ~SCIGCR1_INIT_CLR_VALUE;

    /* [prov] regs.yaml:SCI.SCIPIO0 */
    /* Step 3: Configure pin functions (TX_FUNC, RX_FUNC) */
    REG32(SCIPIO0) |= SCIPIO0_INIT_SET_VALUE;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    /* Step 4: Set TXENA (enable transmitter) */
    REG32(SCIGCR1) |= SCIGCR1_INIT_SET1_VALUE;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    /* Step 5: Set CONT (continue on suspend) */
    REG32(SCIGCR1) |= SCIGCR1_INIT_SET2_VALUE;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    /* Step 6: Set TXENA and RXENA (enable transmitter and receiver) */
    REG32(SCIGCR1) |= SCIGCR1_INIT_SET3_VALUE;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    /* Step 7: Set SWnRST (bring out of reset) */
    REG32(SCIGCR1) |= SCIGCR1_INIT_SET4_VALUE;
}

void sci_configure_format(uint32_t char_length, sci_parity_t parity, sci_stop_bits_t stop_bits)
{
    uint32_t gcr1_val;
    uint32_t format_val;

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    gcr1_val = REG32(SCIGCR1);

    /* Configure stop bits */
    if (stop_bits == SCI_STOP_BITS_2) {
        gcr1_val |= (1u << SCIGCR1_STOP_BIT);
    } else {
        gcr1_val &= ~(1u << SCIGCR1_STOP_BIT);
    }

    /* Configure parity */
    if (parity == SCI_PARITY_NONE) {
        gcr1_val &= ~(1u << SCIGCR1_PARITY_ENA_BIT);
    } else {
        gcr1_val |= (1u << SCIGCR1_PARITY_ENA_BIT);
        if (parity == SCI_PARITY_ODD) {
            gcr1_val |= (1u << SCIGCR1_PARITY_BIT);
        } else {
            gcr1_val &= ~(1u << SCIGCR1_PARITY_BIT);
        }
    }

    /* [prov] regs.yaml:SCI.SCIGCR1 */
    REG32(SCIGCR1) = gcr1_val;

    /* [prov] regs.yaml:SCI.SCIFORMAT */
    /* Configure character length (bits 0..2) */
    format_val = REG32(SCIFORMAT);
    format_val &= ~(0x7u << SCIFORMAT_CHAR_LSB);
    format_val |= ((char_length - 1u) & 0x7u) << SCIFORMAT_CHAR_LSB;
    REG32(SCIFORMAT) = format_val;
}

void sci_set_baudrate(uint32_t baudrate)
{
    uint32_t vclk_hz;
    uint32_t p;
    uint32_t brs_val;

    vclk_hz = clock_get_hz(CLOCKREF_VCLK);

    /* Compute prescaler P for desired baud rate.
     * baud = VCLK / ((P+1) * 16)
     * => P = (VCLK / (baud * 16)) - 1
     * Simplified: P = (vclk_hz / (baudrate * 16)) - 1
     */
    p = (vclk_hz / (baudrate * 16u)) - 1u;

    /* [prov] regs.yaml:SCI.BRS */
    /* Write prescaler P to bits 0..23, M=0, U=0 for simplicity */
    brs_val = (p & 0x00FFFFFFu);
    REG32(BRS) = brs_val;
}

int sci_write_byte(uint8_t data)
{
    /* Wait for TXRDY */
    if (sci_wait_flag(SCIFLR_TXRDY_BIT, SCI_TIMEOUT) != 0) {
        return -1;
    }

    /* [prov] regs.yaml:SCI.SCITD */
    REG32(SCITD) = (uint32_t)data;

    /* Wait for TX_EMPTY to confirm transmission */
    if (sci_wait_flag(SCIFLR_TX_EMPTY_BIT, SCI_TIMEOUT) != 0) {
        return -1;
    }

    return 0;
}

int sci_read_byte(uint8_t *data)
{
    uint32_t val;

    if (data == (void *)0) {
        return -1;
    }

    /* Wait for RXRDY */
    if (sci_wait_flag(SCIFLR_RXRDY_BIT, SCI_TIMEOUT) != 0) {
        return -1;
    }

    /* [prov] regs.yaml:SCI.SCIRD */
    val = REG32(SCIRD);
    *data = (uint8_t)(val & 0xFFu);

    return 0;
}

int sci_write(const uint8_t *buf, uint32_t len)
{
    uint32_t i;
    int ret;

    if (buf == (void *)0) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        ret = sci_write_byte(buf[i]);
        if (ret != 0) {
            return (int)i;
        }
    }

    return (int)len;
}

int sci_read(uint8_t *buf, uint32_t len)
{
    uint32_t i;
    int ret;

    if (buf == (void *)0) {
        return -1;
    }

    for (i = 0; i < len; i++) {
        ret = sci_read_byte(&buf[i]);
        if (ret != 0) {
            return (int)i;
        }
    }

    return (int)len;
}

int sci_tx_ready(void)

SCI (Serial Communication Interface) Driver for TI RM46 (Cortex-R4)
=====================================================================

This driver provides a complete implementation of the SCI/LIN peripheral,
including:

- Initialization and configuration (frame format, baud rate)
- Blocking transmit and receive APIs with timeout
- Interrupt support for level 0 and level 1 IRQs via VIM
- Loopback mode (internal digital and IODFT analog loopback)
- Loopback self-test function for diagnostic purposes
- Status flag reading and clearing
- Example code for polling, interrupt-driven, and loopback modes

USAGE
-----
1. Call sci_init() to initialize the module (enables VCLK, resets, configures pins).
2. Call sci_configure_format() to set character length, parity, and stop bits.
3. Call sci_set_baudrate() to configure the baud rate (uses clock service to get VCLK).
4. Enable TX and RX using sci_enable_tx() and sci_enable_rx().
5. Use sci_write_byte() / sci_read_byte() or sci_write() / sci_read() for data transfer.
6. Optionally, use interrupt APIs to register handlers and enable IRQs.
7. For testing, use sci_loopback_test() to verify the data path.

EXAMPLES
--------
- sci_polling_example.c: Demonstrates blocking transmit and receive.
- sci_interrupt_example.c: Demonstrates interrupt-driven echo operation.
- sci_loopback_example.c: Demonstrates internal loopback self-test.

DEPENDENCIES
------------
- clock.h / clock.c: Provides clock_enable() and clock_get_hz().
- vim.h / vim.c: Provides vim_register_isr(), vim_enable_channel(), vim_disable_channel().

NOTES
-----
- All blocking APIs include bounded timeouts to prevent infinite loops.
- Loopback modes are intended for diagnostic and testing purposes only.
- The driver supports both SCI and LIN modes; LIN-specific features are exposed
  via dedicated registers (LINCOMPARE, LINRD0, LINRD1, LINTD0, LINTD1, etc.).

For more information, refer to the TI RM46L852 Technical Reference Manual.

===== FILE: TESTING.txt =====
SCI Driver Testing Notes
=========================

LOOPBACK SELF-TEST
------------------
The sci_loopback_test() function performs a simple self-test by:
1. Enabling internal loopback mode (LOOP_BACK bit in SCIGCR1).
2. Transmitting a 4-byte test pattern (0x55, 0xAA, 0x0F, 0xF0).
3. Receiving the same pattern back via the loopback path.
4. Verifying that all bytes match.
5. Restoring the original loopback state.

If any step fails (timeout or data mismatch), the function returns -1.

IODFT ANALOG LOOPBACK
---------------------
The sci_enable_iodft_loopback() function enables the LPB_ENA bit in IODFTCTRL,
which routes the TX signal through the physical pins back to RX. This is useful
for testing the analog driver and receiver paths, but requires external pin
connections or a suitable test fixture.

POLLING MODE
------------
The sci_write() and sci_read() functions use bounded timeouts (SCI_TIMEOUT = 100000 iterations)
to prevent infinite loops. In a real system, adjust this value based on the CPU clock and
expected baud rate.

INTERRUPT MODE
--------------
The driver supports two interrupt levels (level 0 and level 1) via the VIM.
Use sci_register_isr_lvl0() and sci_register_isr_lvl1() to install handlers,
then enable the desired interrupts using sci_enable_interrupt() and sci_enable_irq_lvl0()
or sci_enable_irq_lvl1().

The SCIINTVECT0 and SCIINTVECT1 registers provide vector offsets for ISR identification
(see x-ext.interrupts.vect_offsets in soc.yaml). The driver reads these in the ISR if needed.

For production code, replace the simple echo example with application-specific logic.

ERROR HANDLING
--------------
The driver exposes sci_get_flags() and sci_clear_flags() to read and clear error flags
(framing error, overrun error, parity error, etc.). Check SCIFLR after each operation
and clear error flags as appropriate.

BAUD RATE CALCULATION
---------------------
The sci_set_baudrate() function computes the prescaler P using:
    P = (VCLK_Hz / (baudrate * 16)) - 1

For fractional baud rates, adjust the M and U fields in the BRS register. The current
implementation uses M=0 and U=0 for simplicity.

LIN MODE
--------
The driver provides access to LIN-specific registers (LINCOMPARE, LINMASK, LINID, etc.).
To use LIN mode, set the LIN_MODE bit in SCIGCR1 and configure the LIN-specific parameters
as required by the LIN specification. Refer to the RM46 Technical Reference Manual for details.

===== FILE: API_REFERENCE.txt =====
SCI Driver API Reference
=========================

INITIALIZATION
--------------
void sci_init(void)
  - Enables VCLK clock via clock_enable(CLOCKREF_VCLK).
  - Resets the SCI module (SCIGCR0.RESET).
  - Configures pin functions (SCIPIO0).
  - Brings the module out of reset (SCIGCR1.SWnRST).
  - Must be called first before any other SCI functions.

CONFIGURATION
-------------
void sci_configure_format(uint32_t char_length, sci_parity_t parity, sci_stop_bits_t stop_bits)
  - Sets character length (1-8 bits), parity (none/odd/even), and stop bits (1 or 2).
  - Writes to SCIGCR1 and SCIFORMAT registers.

void sci_set_baudrate(uint32_t baudrate)
  - Computes prescaler P = (VCLK_Hz / (baudrate * 16)) - 1.
  - Writes to BRS register.
  - Requires clock service to retrieve VCLK frequency.

ENABLE / DISABLE
----------------
void sci_enable_tx(void)
void sci_disable_tx(void)
void sci_enable_rx(void)
void sci_disable_rx(void)
  - Enable or disable transmitter and receiver (SCIGCR1.TXENA, SCIGCR1.RXENA).

DATA TRANSFER (BLOCKING)
-------------------------
int sci_write_byte(uint8_t data)
  - Transmits a single byte (blocking with timeout).
  - Returns 0 on success, -1 on timeout.
  - Waits for TXRDY flag, writes to SCITD, waits for TX_EMPTY.

int sci_read_byte(uint8_t *data)
  - Receives a single byte (blocking with timeout).
  - Returns 0 on success, -1 on timeout or null pointer.
  - Waits for RXRDY flag, reads from SCIRD.

int sci_write(const uint8_t *buf, uint32_t len)
  - Transmits multiple bytes (blocking).
  - Returns number of bytes successfully transmitted (0..len), or -1 on error.

int sci_read(uint8_t *buf, uint32_t len)
  - Receives multiple bytes (blocking).
  - Returns number of bytes successfully received (0..len), or -1 on error.

STATUS
------
int sci_tx_ready(void)
  - Returns 1 if TXRDY is set, 0 otherwise.

int sci_rx_ready(void)
  - Returns 1 if RXRDY is set, 0 otherwise.

int sci_tx_empty(void)
  - Returns 1 if TX_EMPTY is set, 0 otherwise.

int sci_is_busy(void)
  - Returns 1 if BUSY flag is set, 0 otherwise.

uint32_t sci_get_flags(void)
  - Returns the current value of SCIFLR register.

void sci_clear_flags(uint32_t flags)
  - Clears specified error flags in SCIFLR (write 1 to clear).

LOOPBACK / TEST
---------------
void sci_enable_loopback(void)
void sci_disable_loopback(void)
  - Enable/disable internal digital loopback (SCIGCR1.LOOP_BACK).

void sci_enable_iodft_loopback(void)
void sci_disable_iodft_loopback(void)
  - Enable/disable IODFT analog loopback (IODFTCTRL.LPB_ENA).

int sci_loopback_test(void)
  - Performs automated loopback self-test.
  - Transmits 4-byte pattern, receives it back, verifies.
  - Returns 0 on success, -1 on failure (timeout or mismatch).

INTERRUPT SUPPORT
-----------------
void sci_enable_interrupt(uint32_t int_flag)
void sci_disable_interrupt(uint32_t int_flag)
  - Enable/disable specific interrupt (writes to SCISETINT/SCICLEARINT).
  - int_flag is the bit position (e.g. 9 for RX_INT, 8 for TX_INT).

void sci_set_interrupt_level(uint32_t int_flag, uint32_t level)
  - Set interrupt to level 0 or level 1 (writes to SCISETINTLVL/SCICLEARINTLVL).
  - level = 0 for level 0, 1 for level 1.

int sci_register_isr_lvl0(sci_isr_t isr)
int sci_register_isr_lvl1(sci_isr_t isr)
  - Registers ISR callback for SCI level 0 or level 1 interrupt.
  - Calls vim_register_isr() with VIM channel 64 (lvl0) or 74 (lvl1).
  - Returns 0 on success, non-zero on failure.

int sci_enable_irq_lvl0(void)
int sci_enable_irq_lvl1(void)
int sci_disable_irq_lvl0(void)
int sci_disable_irq_lvl1(void)
  - Enable/disable SCI interrupt channels in VIM.
  - Calls vim_enable_channel() or vim_disable_channel().
  - Returns 0 on success, non-zero on failure.

TYPES
-----
typedef void (*sci_isr_t)(void)
  - ISR callback type.

typedef enum sci_parity_t
  - SCI_PARITY_NONE = 0
  - SCI_PARITY_ODD  = 1
  - SCI_PARITY_EVEN = 2

typedef enum sci_stop_bits_t
  - SCI_STOP_BITS_1 = 0
  - SCI_STOP_BITS_2 = 1

CONSTANTS
---------
SCI_BASE = 0xFFF7E400
SCI_TIMEOUT = 100000 (polling timeout iterations)

IRQ_SCI_LVL0_ID = 64 (VIM channel for SCI level 0)
IRQ_SCI_LVL1_ID = 74 (VIM channel for SCI level 1)
IRQ_LIN_LVL0_ID = 13 (VIM channel for LIN level 0)
IRQ_LIN_LVL1_ID = 27 (VIM channel for LIN level 1)

TYPICAL USAGE SEQUENCE
----------------------
1. sci_init()
2. sci_configure_format(8, SCI_PARITY_NONE, SCI_STOP_BITS_1)
3. sci_set_baudrate(115200)
4. sci_enable_tx()
5. sci_enable_rx()
6. sci_write_byte('A') / sci_read_byte(&ch)

For interrupt-driven:
7. sci_register_isr_lvl0(my_isr)
8. sci_set_interrupt_level(9, 0) // RX_INT to level 0
9. sci_enable_interrupt(9)
10. sci_enable_irq_lvl0()

For loopback test:
7. sci_loopback_test()

SCI Driver Example Scenarios
=============================

EXAMPLE 1: BASIC POLLING TRANSMIT/RECEIVE
------------------------------------------
See: sci_polling_example.c

1. Initialize SCI (sci_init)
2. Configure format: 8 bits, no parity, 1 stop bit
3. Set baud rate to 9600
4. Enable TX and RX
5. Transmit "Hello" (5 bytes)
6. Receive 5 bytes (assuming external loopback or another device)
7. Verify received data matches transmitted data

EXAMPLE 2: INTERRUPT-DRIVEN ECHO
---------------------------------
See: sci_interrupt_example.c

1. Initialize SCI
2. Configure format and baud rate
3. Enable TX and RX
4. Register ISR for level 0 (receive interrupt)
5. Enable RX interrupt and set to level 0
6. Enable VIM channel for level 0
7. In ISR:
   - Check if RXRDY is set
   - Read byte from SCIRD
   - Echo byte back via SCITD
   - Clear RXRDY flag

EXAMPLE 3: LOOPBACK SELF-TEST
------------------------------
See: sci_loopback_example.c

1. Initialize SCI
2. Configure format and baud rate
3. Enable TX and RX
4. Call sci_loopback_test()
   - Function enables internal loopback
   - Transmits test pattern (0x55, 0xAA, 0x0F, 0xF0)
   - Receives data back
   - Verifies data matches
   - Disables loopback
5. Check return value (0 = success, -1 = failure)

EXAMPLE 4: CUSTOM BAUD RATE CALCULATION
----------------------------------------
If the standard sci_set_baudrate() function does not meet your needs,
you can manually compute and write the BRS register:

  uint32_t vclk_hz = clock_get_hz(CLOCKREF_VCLK);
  uint32_t desired_baudrate = 9600;
  uint32_t p = (vclk_hz / (desired_baudrate * 16)) - 1;
  REG32(BRS) = (p & 0x00FFFFFF);

For fractional baud rates, compute M and U fields:
  uint32_t brs_val = ((u & 0x7) << 28) | ((m & 0xF) << 24) | (p & 0xFFFFFF);
  REG32(BRS) = brs_val;

EXAMPLE 5: ERROR HANDLING
--------------------------
After each receive operation, check for errors:

  uint32_t flags = sci_get_flags();
  if (flags & (1u << SCIFLR_PE_BIT)) {
      // Parity error detected
      sci_clear_flags(1u << SCIFLR_PE_BIT);
  }
  if (flags & (1u << SCIFLR_OE_BIT)) {
      // Overrun error detected
      sci_clear_flags(1u << SCIFLR_OE_BIT);
  }
  if (flags & (1u << SCIFLR_FE_BIT)) {
      // Framing error detected
      sci_clear_flags(1u << SCIFLR_FE_BIT);
  }

EXAMPLE 6: IODFT ANALOG LOOPBACK
---------------------------------
To test the analog TX/RX paths:

1. Connect TX pin to RX pin externally (wire loopback)
2. sci_enable_iodft_loopback()
3. sci_write_byte('T')
4. sci_read_byte(&ch)
5. Verify ch == 'T'
6. sci_disable_iodft_loopback()

This tests the full analog signal path including drivers and receivers.

EXAMPLE 7: LIN MODE (ADVANCED)
-------------------------------
To use LIN mode (not fully tested in this driver):

1. sci_init()
2. Set SCIGCR1.LIN_MODE bit
3. Configure LINCOMPARE, LINMASK, LINID registers
4. Use LINTD0/LINTD1 for multi-byte transmit buffers
5. Use LINRD0/LINRD1 for multi-byte receive buffers
