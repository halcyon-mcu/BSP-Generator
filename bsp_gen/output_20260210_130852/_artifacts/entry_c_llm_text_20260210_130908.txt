/**
 * @file entry.c
 * @brief C-level reset handler for TI RM46 MCU startup.
 *
 * This module implements the C-level reset handler invoked by the assembly
 * startup code. It initializes the .data and .bss sections, calls system
 * initialization, and transfers control to main().
 */

#include <stdint.h>
#include <stddef.h>
#include <string.h>

/* Linker-defined symbols marking section boundaries */
extern uint32_t start_of_data;
extern uint32_t end_of_data;
extern uint32_t start_of_data_in_flash;
extern uint32_t start_of_bss;
extern uint32_t end_of_bss;

/* User-provided functions */
extern void system_init(void);
extern int main(void);

/**
 * @brief C-level reset handler.
 * @ingroup BSP_SYSTEM
 *
 * This function is called by the assembly Reset_Handler immediately after
 * the CPU comes out of reset. It performs the following initialization
 * sequence:
 *
 * 1. Copies the .data section from flash to RAM.
 * 2. Zero-initializes the .bss section.
 * 3. Calls system_init() to configure clocks and peripherals.
 * 4. Calls main() to start the application.
 * 5. Enters an infinite loop if main() returns.
 *
 * @note This function is written in ISO C90 to ensure compatibility with
 *       the TI ARM CGT toolchain defaults.
 * @warning Do not call this function directly; it is invoked by the startup
 *          assembly code.
 */
void Reset_Handler_C(void)
{
    size_t data_size;
    size_t bss_size;

    /* Initialize .data section by copying from flash to RAM */
    data_size = (size_t)(&end_of_data - &start_of_data) * sizeof(uint32_t);
    memcpy(&start_of_data, &start_of_data_in_flash, data_size);

    /* Zero-initialize .bss section */
    bss_size = (size_t)(&end_of_bss - &start_of_bss) * sizeof(uint32_t);
    memset(&start_of_bss, 0, bss_size);

    /* Initialize system clocks and peripherals */
    system_init();

    /* Transfer control to application main */
    main();

    /* If main returns, loop forever to avoid undefined behavior */
    for (;;)
    {
        /* Intentionally empty */
    }
}
