108void APP_INTENT_Step(void)
109{
110 uint8_t rx_byte;
111 lin_status_t status;
112
113 /* Guard: only run if initialized */
114 if (s_app_state.magic != APP_MAGIC_INITIALIZED) {
115 return;
116 }
117
118 /* Non-blocking check for UART input */
119 if (!LIN_IsRxReady()) {
120 return;

122
123 /* Read incoming character */
124 status = LIN_ReceiveByte(&rx_byte);
125 if (status != LIN_STATUS_OK) {
126 return;
127 }
128
129 /* Process command */
130 if (rx_byte == 'A' || rx_byte == 'a') {
131 toggle_led_a();
132 print_string("LED A toggled.\r\n");
133 print_prompt();
134 } else if (rx_byte == 'B' || rx_byte == 'b') {
135 toggle_led_b();
136 print_string("LED B toggled.\r\n");
137 print_prompt();
138 } else {
139 print_string("Unknown command. Press 'A' to toggle LED A, 'B' to toggle LED B.\r\n");
140 print_prompt();
141 }
142}
